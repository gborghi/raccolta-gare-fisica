\begin{tikzpicture}[>=Latex,xscale=1.05,yscale=1.4]
  % axes
  \draw[->] (-3.6,0) -- (3.7,0) node[right] {$x_1$};
  \draw[->] (0,0) -- (0,2.4) node[above] {$x_2$};
  % x ticks
  \foreach \x in {-3,-2,-1,1,2,3} {
    \draw (\x,0.05) -- (\x,-0.05) node[below] {$\x$};
  }
  % y ticks
  \foreach \y in {0.5,1,1.5,2} {
    \draw (0.05,\y) -- (-0.05,\y) node[left] {$\y$};
  }
  % symmetric U-shaped curve
  \draw[domain=-3.1:3.1,samples=120,smooth]
    plot (\x,{2*(abs(\x)/3)^4});
  % point marker on the curve (right branch, x = 1.5)
  \fill (1.5,0.125) circle (1.8pt);
\end{tikzpicture}
