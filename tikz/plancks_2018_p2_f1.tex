\begin{tikzpicture}[>=Latex,scale=1.0]
  % conjugated polyene chain (zigzag backbone of C atoms)
  % backbone vertices
  \coordinate (c1) at (1.0,0);
  \coordinate (c2) at (2.0,0.55);
  \coordinate (c3) at (3.0,0);
  \coordinate (c4) at (4.0,0.55);

  \foreach \p in {c1,c3,c5,c7} {\node[fill=white,inner sep=1pt] at (\p) {$C$};}
  \foreach \p in {c2,c4,c6,c8} {\node[fill=white,inner sep=1pt] at (\p) {$C$};}
  % single bonds along backbone
  \draw (c2) -- (c3);

  \draw (c1) -- (c2);

  \draw (c1) -- (0.1,0.55);

  \draw (c8) -- (8.9,0);

  % H atoms on bottom carbons
  \foreach \p in {c1,c3,c5,c7} {\draw (\p) -- ($(\p)+(0,-0.7)$); \node at ($(\p)+(0,-0.95)$) {$H$};}
